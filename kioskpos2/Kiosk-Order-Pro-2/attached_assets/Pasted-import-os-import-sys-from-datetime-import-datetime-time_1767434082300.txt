# Android file paths

def get_android_storage_path():
    """Get writable storage path on Android"""
    try:
        from android.storage import app_storage_path
        from jnius import autoclass
        Environment = autoclass('android.os.Environment')
        return Environment.getExternalStorageDirectory().getAbsolutePath()
    except:
        return os.path.dirname(os.path.abspath(__file__))

def get_reports_path():
    """Get path for saving reports"""
    base_path = get_android_storage_path()
    reports_path = os.path.join(base_path, "KioskPOS_Reports")
    os.makedirs(reports_path, exist_ok=True)
    return reports_path

def load_settings():
    settings_path = os.path.join(get_android_storage_path(), SETTINGS_FILE)
    if os.path.exists(settings_path):
        try:
            with open(settings_path, "r", encoding="utf-8") as f:
                return json.load(f)
        except:
            pass
    # Default settings now just uses 'EN'
    return {"lang": "EN", "shop_name": "Kiosk POS System"}

def save_settings(s):
    settings_path = os.path.join(get_android_storage_path(), SETTINGS_FILE)
    with open(settings_path, "w", encoding="utf-8") as f:
        json.dump(s, f, ensure_ascii=False, indent=2)

def ensure_db():
    db_path = os.path.join(get_android_storage_path(), DB_FILE)
    if not os.path.exists(db_path):
        conn = sqlite3.connect(db_path)
        c = conn.cursor()
        c.execute("""CREATE TABLE menu (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT UNIQUE NOT NULL, price REAL NOT NULL)""")
        c.execute("""CREATE TABLE sales (id INTEGER PRIMARY KEY AUTOINCREMENT, date TEXT NOT NULL, item_name TEXT NOT NULL, quantity INTEGER NOT NULL, price REAL NOT NULL, payment_mode TEXT NOT NULL, notes TEXT)""")
        c.executemany("INSERT INTO menu (name,price) VALUES (?,?)", [('Burger',50), ('Tea',10), ('Coffee',20), ('Sandwich',40)])
        conn.commit()
        conn.close()

ensure_db()

def db_conn():
    db_path = os.path.join(get_android_storage_path(), DB_FILE)
    return sqlite3.connect(db_path)

def get_menu_items():
    conn = db_conn()
    c = conn.cursor()
    c.execute("SELECT id, name, price FROM menu ORDER BY name")
    rows = c.fetchall()
    conn.close()
    return rows

def add_menu_item(name, price):
    conn = db_conn()
    c = conn.cursor()
    try:
        c.execute("INSERT INTO menu (name,price) VALUES (?,?)", (name, price))
        conn.commit()
        return True
    except sqlite3.IntegrityError:
        return False
    finally:
        conn.close()

def update_menu_item(item_id, name, price):
    conn = db_conn()
    c = conn.cursor()
    try:
        c.execute("UPDATE menu SET name=?, price=? WHERE id=?", (name, price, item_id))
        conn.commit()
        return True
    except:
        return False
    finally:
        conn.close()

def delete_menu_item(item_id):
    conn = db_conn()
    c = conn.cursor()
    try:
        c.execute("DELETE FROM menu WHERE id=?", (item_id,))
        conn.commit()
        return True
    except:
        return False
    finally:
        conn.close()

def record_sale(item_name, quantity, price, payment_mode, notes):
    conn = db_conn()
    c = conn.cursor()
    dt = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    c.execute("INSERT INTO sales (date,item_name,quantity,price,payment_mode,notes) VALUES (?,?,?,?,?,?)", (dt, item_name, quantity, price, payment_mode, notes))
    conn.commit()
    conn.close()

def get_sales_by_date(date_str):
    conn = db_conn()
    c = conn.cursor()
    c.execute("SELECT id,date,item_name,quantity,price,payment_mode,notes FROM sales WHERE date LIKE ? ORDER BY id", (f"{date_str}%",))
    rows = c.fetchall()
    conn.close()
    return rows

class NumberPad(Popup):
    # --- NumberPad CLASS FOR MULTI-DIGIT INPUT ---
    def __init__(self, callback, **kwargs):
        super().__init__(**kwargs)
        self.callback = callback
        self.current_input = ''
        self.title = 'Quantity'
        self.title_font = DEFAULT_FONT
        self.size_hint = (0.8, 0.6)
        self.auto_dismiss = False
        
        main_layout = BoxLayout(orientation='vertical', spacing=5, padding=10)

        # Display for current input
        self.input_label = Label(text='0', font_size=32, size_hint_y=0.2, color=(1,1,1,1))
        main_layout.add_widget(self.input_label)
        
        # Grid for number buttons
        layout = GridLayout(cols=3, rows=4, spacing=5, size_hint_y=0.8)
        numbers = ['7', '8', '9', '4', '5', '6', '1', '2', '3', 'C', '0', 'OK']
        for num in numbers:
            btn = Button(text=num, font_size=24, background_color=get_color_from_hex('#333333'), color=(1,1,1,1))
            btn.bind(on_release=self.on_button_press)
            layout.add_widget(btn)

        main_layout.add_widget(layout)
        self.content = main_layout

    def on_button_press(self, instance):
        text = instance.text
        if text == 'OK':
            if self.current_input.isdigit() and int(self.current_input) > 0:
                self.callback(self.current_input)
            else:
                self.callback('') # Send empty string for no action/invalid input
            self.dismiss()
        elif text == 'C':
            self.current_input = ''
            self.input_label.text = '0'
        elif text.isdigit():
            # Max length check to prevent absurdly large numbers
            if len(self.current_input) < 5:
                self.current_input += text
                self.input_label.text = self.current_input

    # --- Function to filter and display menu items ---
    def display_menu_items(self, menu_items):
        self.menu_layout.clear_widgets()
        for item_id, name, price in menu_items:
            item_btn = Button(
                text=f"{name}\n₹{price:.2f}",
                size_hint_y=None,
                height=120,
                background_color=get_color_from_hex('#444444'),
                color=(1,1,1,1),
                font_size='16sp',
                font_name=DEFAULT_FONT
            )
            item_btn.bind(on_release=lambda x, n=name, p=price: self.add_to_cart(n, p))
            self.menu_layout.add_widget(item_btn)

    def filter_menu_items(self, instance, value):
        search_term = value.lower()
        # Refresh menu items in case of changes in the menu screen
        self.all_menu_items = get_menu_items() 
        
        filtered_items = [
            item for item in self.all_menu_items 
            if search_term in item[1].lower() # item[1] is the name
        ]
        self.display_menu_items(filtered_items)

    def add_to_cart(self, name, price):
        def on_quantity_selected(value):
            if value and value.isdigit():
                qty = int(value)
                if qty > 0:
                    # Check if item already in cart
                    for item in self.cart:
                        if item['name'] == name:
                            item['quantity'] += qty
                            item['price'] = item['unit_price'] * item['quantity']
                            self.update_cart_display()
                            return

                    # Add new item
                    self.cart.append({
                        'name': name,
                        'quantity': qty,
                        'price': price * qty,
                        'unit_price': price
                    })
                    self.update_cart_display()

        numpad = NumberPad(on_quantity_selected)
        numpad.open()

    def update_cart_display(self):
        self.cart_layout.clear_widgets()
        total_amount = 0

        for item in self.cart:
            cart_item = CartItem(
                item['name'],
                item['quantity'],
                item['price'],
                self.remove_from_cart
            )
            self.cart_layout.add_widget(cart_item)
            total_amount += item['price']

        # Add total row
        if self.cart:
            total_layout = BoxLayout(orientation='horizontal', size_hint_y=None, height=50)
            total_layout.add_widget(Label(text="Total:", font_size='16sp', bold=True, color=(1,1,1,1), font_name=DEFAULT_FONT))
            total_layout.add_widget(Label(text=f"₹{total_amount:.2f}", font_size='16sp', bold=True, color=(1,1,1,1), font_name=DEFAULT_FONT))
            total_layout.add_widget(Label(text="", size_hint_x=0.1))  # spacer
            self.cart_layout.add_widget(total_layout)

    def remove_from_cart(self, name):
        self.cart = [item for item in self.cart if item['name'] != name]
        self.update_cart_display()

    def clear_cart(self, instance):
        self.cart = []
        self.update_cart_display()
        self.show_alert("Cart Cleared!")

    def checkout(self, instance):
        if not self.cart:
            self.show_alert("Cart Empty")
            return

        content = BoxLayout(orientation='vertical', spacing=10, padding=20)

        content.add_widget(Label(text='Payment Mode', font_size='18sp', color=(1,1,1,1), font_name=DEFAULT_FONT))
        payment_spinner = Spinner(
            text='Cash',
            values=('Cash', 'UPI', 'Card'),
            size_hint_y=0.2,
            background_color=get_color_from_hex('#333333'),
            color=(1,1,1,1)
        )
        content.add_widget(payment_spinner)

        content.add_widget(Label(text='Notes', font_size='18sp', color=(1,1,1,1), font_name=DEFAULT_FONT))
        notes_input = TextInput(hint_text='Notes', multiline=True, size_hint_y=0.3,
                              background_color=get_color_from_hex('#333333'), foreground_color=(1,1,1,1), font_name=DEFAULT_FONT)
        content.add_widget(notes_input)

        total = sum(item['price'] for item in self.cart)
        total_label = Label(text=f"Total: ₹{total:.2f}", font_size='20sp', bold=True, color=(1,1,1,1), font_name=DEFAULT_FONT)
        content.add_widget(total_label)

        def process_checkout(btn):
            # Record sales
            for item in self.cart:
                record_sale(
                    item['name'],
                    item['quantity'],
                    item['unit_price'] * item['quantity'],
                    payment_spinner.text,
                    notes_input.text
                )

            self.send_chef_message(notes_input.text)

            self.cart = []
            if self.popup:
                self.popup.dismiss()
            self.show_alert("Order Complete!")
            self.go_back()

        btn_layout = BoxLayout(size_hint_y=0.2, spacing=10)
        confirm_btn = Button(text='Confirm', background_color=get_color_from_hex('#4CAF50'))
        cancel_btn = Button(text='Cancel', background_color=get_color_from_hex('#F44336'), font_name=DEFAULT_FONT)

        confirm_btn.bind(on_release=process_checkout)
        cancel_btn.bind(on_release=lambda x: self.popup.dismiss() if self.popup else None)

        btn_layout.add_widget(confirm_btn)
        btn_layout.add_widget(cancel_btn)
        content.add_widget(btn_layout)

        self.show_popup('Checkout', content)

    def send_chef_message(self, notes):
        """WhatsApp integration"""
        try:
            grouped = Counter()
            for item in self.cart:
                grouped[item['name']] += item['quantity']

            message = "New Order:\n"

            for item, qty in grouped.items():
                message += f"{item} x{qty}\n"

            message += f"\nTotal: ₹{sum(item['price'] for item in self.cart):.2f}"
            if notes:
                message += f"\nNotes: {notes}"

            # Create WhatsApp URL
            import urllib.parse
            encoded_message = urllib.parse.quote(message)
            whatsapp_url = f"https://wa.me/{CHEF_NUMBER}?text={encoded_message}"

            # Try to open WhatsApp
            try:
                import webbrowser
                webbrowser.open(whatsapp_url)
            except:
                self.show_alert(f"Message ready for WhatsApp:\n\n{message}")

        except Exception as e:
            self.show_alert(f"Error: {str(e)}")

        cancel_btn = Button(text='Cancel', background_color=get_color_from_hex('#F44336'), font_name=DEFAULT_FONT)

        cancel_btn.bind(on_release=lambda x: self.popup.dismiss() if self.popup else None)

        btn_layout.add_widget(save_btn)
        btn_layout.add_widget(cancel_btn)
        content.add_widget(btn_layout)

    def show_report(self, date_str):

        sales = get_sales_by_date(date_str)

        content = BoxLayout(orientation='vertical', spacing=10, padding=10)

        # Back button
        back_btn = Button(text='Back', size_hint_y=0.08, background_color=get_color_from_hex('#666666'), font_name=DEFAULT_FONT)

        content.add_widget(back_btn)

            content.add_widget(Label(text="No sales for this date", font_size='18sp', color=(1,1,1,1), font_name=DEFAULT_FONT))

        total = sum(s[4] for s in sales)
        total_label = Label(text=f"Total Sales: ₹{total:.2f}", font_size='20sp', bold=True, color=(1,1,1,1), font_name=DEFAULT_FONT)
        content.add_widget(total_label)

        # Detailed sales table (now taking up more vertical space)
        table_label = Label(text="Detailed Sales:", size_hint_y=0.05, font_size='16sp', bold=True, color=(1,1,1,1), font_name=DEFAULT_FONT)
        content.add_widget(table_label)

        # Increased size_hint_y to 0.75 to occupy the space where the graph was
        scroll = ScrollView(size_hint_y=0.75)
        # Use a consistent height for row to make it look like a proper table
        report_layout = GridLayout(cols=4, spacing=5, size_hint_y=None, row_default_height=40, row_force_default=True)
        report_layout.bind(minimum_height=report_layout.setter('height'))

        # Header
        headers = ['Time', 'Item', 'Quantity', 'Price']
        for header in headers:
            lbl = Label(text=header, bold=True, font_size='14sp', color=get_color_from_hex('#29B6F6'), font_name=DEFAULT_FONT)
            report_layout.add_widget(lbl)

        # Data rows
        for sale in sales:
            time = sale[1].split(' ')[1][:5]  # HH:MM format
            report_layout.add_widget(Label(text=time, font_size='12sp', color=(1,1,1,1), font_name=DEFAULT_FONT))
            report_layout.add_widget(Label(text=sale[2], font_size='12sp', color=(1,1,1,1), font_name=DEFAULT_FONT))
            report_layout.add_widget(Label(text=str(sale[3]), font_size='12sp', color=(1,1,1,1), font_name=DEFAULT_FONT))
            report_layout.add_widget(Label(text=f"₹{sale[4]:.2f}", font_size='12sp', color=(1,1,1,1), font_name=DEFAULT_FONT))

        scroll.add_widget(report_layout)
        content.add_widget(scroll)

        csv_btn = Button(text="Export as CSV", size_hint_y=0.07, background_color=get_color_from_hex('#2196F3'))
        csv_btn.bind(on_release=lambda x: self.export_report_as_csv(date_str, sales))
        content.add_widget(csv_btn)

        self.show_popup(f"Sales Report - {date_str}", content, size_hint=(0.95, 0.95))

    def export_report_as_csv(self, date_str, sales):
        """Export report as CSV for opening in spreadsheet editor"""
        try:
            reports_path = get_reports_path()
            filename = os.path.join(reports_path, f"sales_report_{date_str}.csv")

            with open(filename, 'w', encoding='utf-8', newline='') as csvfile:
                fieldnames = ['ID', 'Date', 'Time', 'Item_Name', 'Quantity', 'Total_Price', 'Payment_Mode', 'Notes']
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames)

                writer.writeheader()
                for sale in sales:
                    sale_date_time = datetime.strptime(sale[1], "%Y-%m-%d %H:%M:%S")
                    writer.writerow({
                        'ID': sale[0],
                        'Date': sale_date_time.strftime("%Y-%m-%d"),
                        'Time': sale_date_time.strftime("%H:%M:%S"),
                        'Item_Name': sale[2],
                        'Quantity': sale[3],
                        'Total_Price': f"₹{sale[4]:.2f}",
                        'Payment_Mode': sale[5],
                        'Notes': sale[6]
                    })

            self.show_alert(f"Report exported successfully!\n\nFile: {os.path.basename(filename)}\nPath: {reports_path}\n\nPlease use a file manager app to open the CSV.")

        except Exception as e:
            self.show_alert(f"Error exporting report: {str(e)}")
